// SERVER-SIDE M3U PARSER

function parseM3UServer(m3uText) {
  const lines = m3uText.split(/\r?\n/);
  const channels = [];
  
  // Isolated localized tracking parameters to avoid Cloudflare compilation warnings
  let trackedName = "Unknown";
  let trackedLogo = "";
  let trackedGroup = "Uncategorized";
  let trackedLicense = "";

  for (let i = 0; i < lines.length; i++) {
      const line = lines[i].trim();
      if (!line) continue;

      if (line.startsWith('#EXTINF:')) {
          const tvgNameMatch = line.match(/tvg-name="([^"]+)"/);
          const tvgLogoMatch = line.match(/tvg-logo="([^"]+)"/);
          const groupTitleMatch = line.match(/group-title="([^"]+)"/);
          
          const commaSplit = line.split(',');
          const nameFallback = commaSplit.length > 1 ? commaSplit.pop().trim() : 'Unknown';

          trackedName = tvgNameMatch ? tvgNameMatch[1] : nameFallback;
          trackedLogo = tvgLogoMatch ? tvgLogoMatch[1] : '';
          trackedGroup = groupTitleMatch ? groupTitleMatch[1] : 'Uncategorized';
          trackedLicense = ""; // reset for next entry
      } 
      else if (line.startsWith('#EXT-X-LICENSE-URL:')) {
          trackedLicense = line.replace('#EXT-X-LICENSE-URL:', '').trim();
      } 
      else if (line.startsWith('#KODIPROP:inputstream.adaptive.license_key=')) {
          if (!trackedLicense) {
              trackedLicense = line.replace('#KODIPROP:inputstream.adaptive.license_key=', '').trim();
          }
      } 
      else if (line.startsWith('http')) {
          let rawUrl = line;
          const proxyPrefix = "https://allinone.godess.workers.dev/?url=";
          
          if (rawUrl.startsWith(proxyPrefix)) {
              rawUrl = decodeURIComponent(rawUrl.substring(proxyPrefix.length));
          }

          const parts = rawUrl.split('|');
          const finalMpdUrl = parts[0];
          let userAgent = "";
          let finalCookie = "";

          if (parts.length > 1) {
              const headerParts = parts[1].split('&');
              for (const hp of headerParts) {
                  if (hp.startsWith('User-Agent=')) {
                      userAgent = hp.substring(11); 
                  } else if (hp.startsWith('Cookie=')) {
                      finalCookie = hp.substring(7); 
                  }
              }
          }
          
          channels.push({
              name: trackedName,
              logo: trackedLogo,
              group: trackedGroup,
              license_url: trackedLicense,
              mpd_url: finalMpdUrl,
              user_agent: userAgent,
              headers: { cookie: finalCookie }
          });
      }
  }
  return channels;
}

export default {
  async fetch(request) {
    const url = new URL(request.url);
    const path = url.pathname;

    // FIX: Process specific API routes BEFORE broad template matching checks
    if (path === '/channels') {
      try {
        const DATA_SOURCE = 'https://noisy-truth-6766.streamstar18.workers.dev/';
        const PROXY_URL = 'https://allinone.godess.workers.dev/?url=';
        
        // Route through proxy to prevent scraping restrictions
        //const fetchTarget = eDATA_SOURCE);
        const response = await fetch(DATA_SOURCE);
        
        if (!response.ok) {
            throw new Error(`Proxy returned status: ${response.status}`);
        }

        const rawData = await response.text();
        let channelsData = [];
        
        // Automatically determine format structure and route parse pattern
        if (rawData.trim().startsWith('#EXTM3U')) {
            channelsData = parseM3UServer(rawData);
        } else {
            channelsData = JSON.parse(rawData);
        }

        return new Response(JSON.stringify(channelsData), { 
            headers: { 
                'Content-Type': 'application/json; charset=UTF-8',
                'Access-Control-Allow-Origin': '*'
            } 
        });

      } catch (error) {
        return new Response(JSON.stringify({ error: error.message || "Failed to process channels configuration." }), {
            status: 500,
            headers: { 
                'Content-Type': 'application/json',
                'Access-Control-Allow-Origin': '*'
            }
        });
      }
    }

    // Serve HTML page route
    if (path === '/' || path.startsWith('/c')) {
      return new Response(htmlContent, { headers: { 'Content-Type': 'text/html; charset=UTF-8' } });
    }

    // Default Fallback
    return new Response('Titan Zenith 100% Operational', { status: 200 });
  }
};
