// Pre-allocated hex lookup table for high-speed byte-to-hex translation
const HEX_STRINGS = Array.from({ length: 256 }, (_, i) => i.toString(16).padStart(2, '0'));

export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const targetUrl = url.searchParams.get('url');
    
    // 1. Validate the incoming query parameter
    if (!targetUrl) {
      return new Response('Error: Missing "url" query parameter. Usage: ?url=https://domain.com/key.php', { 
        status: 400,
        headers: { 'Content-Type': 'text/plain; charset=utf-8' }
      });
    }

    // 2. Setup the full spoofed browser stack headers featuring your identity token
    const advancedHeaders = {
      'User-Agent': 'https://t.me/cloudply || @cloudplay',
      'Accept': 'application/json, text/plain, */*',
      'Accept-Language': 'en-US,en;q=0.9',
      'Accept-Encoding': 'gzip, deflate, br',
      'Cache-Control': 'no-cache',
      'Pragma': 'no-cache',
      'Sec-Ch-Ua': '"Chromium";v="122", "Not(A:Brand";v="24"',
      'Sec-Ch-Ua-Mobile': '?0',
      'Sec-Ch-Ua-Platform': '"Windows"',
      'Sec-Fetch-Dest': 'empty',
      'Sec-Fetch-Mode': 'cors',
      'Sec-Fetch-Site': 'same-origin',
      'Upgrade-Insecure-Requests': '1'
    };

    // Forward the Content-Type header if the client provided one
    if (request.headers.has('content-type')) {
      advancedHeaders['Content-Type'] = request.headers.get('content-type');
    }

    // 3. Read incoming body payload only if the client sent one
    let body = null;
    if (request.method !== 'GET' && request.method !== 'HEAD') {
      body = await request.arrayBuffer(); 
    }

    try {
      // 4. Force the network request to ALWAYS use POST
      const response = await fetch(targetUrl, {
        method: 'POST',
        headers: advancedHeaders,
        body: body,
        cf: {
          cacheTtl: -1,          // Instantly forces a fresh fetch from source (No caching latency)
          apps: false,           // Strips out script injection systems
          mirage: false,         // Bypasses optimization routines for imagery
          polish: 'off'          // Deactivates compression pipeline engines
        }
      });

      if (!response.ok) {
        return new Response(`Origin Error: Target returned status ${response.status}`, { status: response.status });
      }

      // 5. Parse JSON payload
      const data = await response.json();

      if (!data || !data.keys || !data.keys[0]) {
        return new Response('Error: Invalid JSON payload format from source target.', { status: 422 });
      }

      const { kid, k } = data.keys[0];

      if (!kid || !k) {
        return new Response('Error: Target JSON payload is missing "kid" or "k" keys.', { status: 422 });
      }

      // 6. High-speed raw byte hex conversions
      const kidHex = fastBase64UrlToHex(kid);
      const kHex = fastBase64UrlToHex(k);

      // 7. Return standard keyid:key output format
      return new Response(`${kidHex}:${kHex}`, {
        headers: { 
          'Content-Type': 'text/plain; charset=utf-8',
          'Access-Control-Allow-Origin': '*',                          // Enables API usability anywhere
          'Cache-Control': 'no-store, no-cache, must-revalidate, max-age=0' // Prevents old data caching
        }
      });

    } catch (error) {
      return new Response(`Execution Exception Error: ${error.message}`, { status: 500 });
    }
  }
};

/**
 * Supercharged conversion helper function.
 * Avoids heavy V8 string concatenation engine operations by processing via prebuilt arrays.
 */
function fastBase64UrlToHex(base64url) {
  // Translate URL-safe Base64 symbols back to standard format
  let base64 = base64url.replace(/-/g, '+').replace(/_/g, '/');
  
  // Re-append standard cryptographic chunk padding if omitted
  while (base64.length % 4) {
    base64 += '=';
  }

  // Deconstruct standard string payload to binary layout
  const binaryString = atob(base64);
  const len = binaryString.length;
  
  // Process hex compilation strings utilizing pre-allocated table indexing references
  let hexResult = '';
  for (let i = 0; i < len; i++) {
    hexResult += HEX_STRINGS[binaryString.charCodeAt(i)];
  }
  
  return hexResult;
}
